━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Check the operator is running:
   kubectl -n {{ .Release.Namespace }} get deploy {{ include "clamav-operator.fullname" . }}-controller-manager

2. Trigger a full cluster scan:
   kubectl apply -f - <<EOF
   apiVersion: clamav.io/v1alpha1
   kind: ClusterScan
   metadata:
     name: quick-scan
     namespace: {{ .Release.Namespace }}
   spec:
     concurrent: 3
   EOF

3. Watch the scan progress:
   kubectl -n {{ .Release.Namespace }} get nodescans -w

4. Check results:
   kubectl -n {{ .Release.Namespace }} get clusterscan quick-scan -o jsonpath='{.status.phase}'
