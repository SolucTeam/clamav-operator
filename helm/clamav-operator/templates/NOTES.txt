1. Verify the operator is running:
   kubectl -n {{ .Release.Namespace }} rollout status deploy \
     {{ include "clamav-operator.fullname" . }}-controller-manager

2. Run the Helm test suite:
   helm test {{ .Release.Name }} -n {{ .Release.Namespace }}

3. Trigger a full cluster scan:
   kubectl apply -f - <<EOF
   apiVersion: clamav.io/v1alpha1
   kind: ClusterScan
   metadata:
     name: quick-scan
     namespace: {{ .Release.Namespace }}
   spec:
     concurrent: 3
   EOF

4. Watch progress:
   kubectl -n {{ .Release.Namespace }} get nodescans -w

5. Check results:
   kubectl -n {{ .Release.Namespace }} get clusterscan quick-scan \
     -o jsonpath='{.status.phase}'
